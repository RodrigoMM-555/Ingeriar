# Roles y permisos (E-13). El instalador carga los valores iniciales de
# server/data/semillas/roles-permisos.sql.
roles
- codigo VARCHAR(50) PRIMARY KEY
- nombre VARCHAR(100) NOT NULL
- basico BOOLEAN NOT NULL DEFAULT FALSE
- activo BOOLEAN NOT NULL DEFAULT TRUE
- orden SMALLINT UNSIGNED NOT NULL DEFAULT 0

permisos
- codigo VARCHAR(60) PRIMARY KEY
- descripcion VARCHAR(255) NOT NULL

usuarios_roles
- id_usuario INT UNSIGNED NOT NULL REFERENCES usuarios(id) ON DELETE CASCADE
- codigo_rol VARCHAR(50) NOT NULL REFERENCES roles(codigo) ON DELETE CASCADE
- PRIMARY KEY (id_usuario, codigo_rol)

roles_permisos
- codigo_rol VARCHAR(50) NOT NULL REFERENCES roles(codigo) ON DELETE CASCADE
- codigo_permiso VARCHAR(60) NOT NULL REFERENCES permisos(codigo) ON DELETE CASCADE
- PRIMARY KEY (codigo_rol, codigo_permiso)

educandos
- nombre VARCHAR(100) NOT NULL
- apellidos VARCHAR(150) NOT NULL
- DNI CHAR(9) NOT NULL UNIQUE
- naz DATE
- fecha_baja DATE NULL

facturas
- responsable_id INT UNSIGNED NOT NULL REFERENCES usuarios(id)
- numero_fact INT NOT NULL
- CIF_prov VARCHAR(20) NOT NULL
- fecha_emision DATETIME NOT NULL
- fecha_compra DATETIME NOT NULL
- concepto TEXT NOT NULL
- precio DECIMAL(10,2) NOT NULL
- categoria enum('envíos', 'comida', 'material', 'gasolina', 'otros', 'agua', 'luz', 'gas') NOT NULL
- estado ENUM('pendiente', 'revisada', 'pagada') NOT NULL DEFAULT 'pendiente'
- imagen_factura VARCHAR(255) NULL
- estado_cambiado_por INT UNSIGNED NULL REFERENCES usuarios(id) ON DELETE SET NULL
- estado_cambiado_en DATETIME NULL

usuarios_educandos
- id_usuario INT UNSIGNED NOT NULL REFERENCES usuarios(id) ON DELETE CASCADE
- id_educando INT UNSIGNED NOT NULL REFERENCES educandos(id) ON DELETE CASCADE
- UNIQUE (id_usuario, id_educando)

eventos
- nombre VARCHAR(150) NOT NULL
- fecha_inicio DATETIME NOT NULL
- fecha_fin DATETIME NOT NULL
- id_responsable INT UNSIGNED REFERENCES usuarios(id) ON DELETE SET NULL
- lugar VARCHAR(150)
- municipio VARCHAR(100)
- comunidad VARCHAR(100)
- tipo VARCHAR(50) NOT NULL
- circular BOOLEAN NOT NULL DEFAULT FALSE

eventos_secciones
- id_evento INT UNSIGNED NOT NULL REFERENCES eventos(id) ON DELETE CASCADE
- seccion VARCHAR(50) NOT NULL
- PRIMARY KEY (id_evento, seccion)

educandos_asistencias
- id_educando INT UNSIGNED NOT NULL REFERENCES educandos(id) ON DELETE CASCADE
- id_evento INT UNSIGNED NOT NULL REFERENCES eventos(id) ON DELETE CASCADE
- estado ENUM('pendiente', 'si', 'no') NOT NULL DEFAULT 'pendiente'
- archivo_entrega VARCHAR(255)
- UNIQUE (id_educando, id_evento)

galeria_fotos
- id_evento INT UNSIGNED NOT NULL REFERENCES eventos(id) ON DELETE CASCADE
- archivo VARCHAR(255) NOT NULL UNIQUE
- nombre_original VARCHAR(255) NOT NULL
- id_usuario INT UNSIGNED NULL REFERENCES usuarios(id) ON DELETE SET NULL
- fecha_subida DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP

registro_accesos
- fecha DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
- id_usuario INT UNSIGNED NULL REFERENCES usuarios(id) ON DELETE SET NULL
- usuario_email VARCHAR(254) NOT NULL
- accion VARCHAR(30) NOT NULL
- recurso VARCHAR(30) NOT NULL
- id_educando INT UNSIGNED NULL REFERENCES educandos(id) ON DELETE SET NULL
- detalle VARCHAR(255) NULL
- ip VARCHAR(45) NULL

# Documentación de inscripción (E-14). El instalador carga los valores iniciales de
# server/data/semillas/documentos-base.sql.
documentos_base
- codigo VARCHAR(50) PRIMARY KEY
- nombre VARCHAR(150) NOT NULL
- tipo ENUM('participacion', 'inscripcion') NOT NULL DEFAULT 'inscripcion'
- archivo_plantilla VARCHAR(255) NULL
- obligatorio BOOLEAN NOT NULL DEFAULT TRUE
- activo BOOLEAN NOT NULL DEFAULT TRUE
- orden SMALLINT UNSIGNED NOT NULL DEFAULT 0

documentos_educando
- id_educando INT UNSIGNED NOT NULL REFERENCES educandos(id) ON DELETE CASCADE
- codigo_documento VARCHAR(50) NOT NULL REFERENCES documentos_base(codigo)
- archivo VARCHAR(255) NOT NULL
- subido_por INT UNSIGNED NULL REFERENCES usuarios(id) ON DELETE SET NULL
- fecha_subida DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
- estado ENUM('pendiente', 'revisado', 'corregir') NOT NULL DEFAULT 'pendiente'
- observaciones VARCHAR(255) NULL
- revisado_por INT UNSIGNED NULL REFERENCES usuarios(id) ON DELETE SET NULL
- revisado_en DATETIME NULL
- UNIQUE (id_educando, codigo_documento)
